cmake_minimum_required(VERSION 3.25)
project(ChartsHubDrumHost VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The drum host currently supports Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "Never install a test plugin" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "No separate VC runtime installer" FORCE)
set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "Offline test needs no module metadata tool" FORCE)
set(SMTG_ADD_VST3_UTILITIES OFF CACHE BOOL "No sample utilities in the shipped helper" FORCE)
add_subdirectory(vendor/vst3sdk EXCLUDE_FROM_ALL)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/vst3sdk")
add_executable(ChartsHubDrumHost src/main.cpp src/vst-host.cpp src/windows-audio.cpp
  "${SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp"
  "${SDK_ROOT}/public.sdk/source/vst/hosting/plugprovider.cpp")
target_include_directories(ChartsHubDrumHost PRIVATE src vendor/json "${SDK_ROOT}")
target_compile_definitions(ChartsHubDrumHost PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(ChartsHubDrumHost PRIVATE /W4 /EHsc /utf-8 /fp:precise)
target_link_libraries(ChartsHubDrumHost PRIVATE sdk_hosting ole32 uuid winmm propsys avrt user32 shell32)
set_target_properties(ChartsHubDrumHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(ChartsHubDrumHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/bin")
endforeach()
add_custom_command(TARGET ChartsHubDrumHost POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "${CMAKE_CURRENT_SOURCE_DIR}/bin/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/THIRD-PARTY-NOTICES.txt" "${CMAKE_CURRENT_SOURCE_DIR}/bin/THIRD-PARTY-NOTICES.txt")
option(CHARTSHUB_DRUM_TEST_PLUGIN "Build the offline-only synthetic instrument" ON)
if(CHARTSHUB_DRUM_TEST_PLUGIN)
  smtg_add_vst3plugin(ChartsHubDrumTest tests/synthetic-plugin.cpp)
  target_link_libraries(ChartsHubDrumTest PRIVATE sdk)
  target_compile_definitions(ChartsHubDrumTest PRIVATE NOMINMAX)
endif()
